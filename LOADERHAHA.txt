if not game:IsLoaded() then
    game.Loaded:Wait()
end

-- Services
local Players = game:GetService("Players")

-- Prevent multiple executions
if _G.NocytraHubLoaded then
    return
end
_G.NocytraHubLoaded = true

-- Configuration: Game PlaceId to Script Mapping
local GameScripts = {
    [129866685202296] = "https://pastebin.com/raw/tLQhkWSm",
    [125810438250765] = "https://pastebin.com/raw/9tcVqgDm",
    [93044798454681] = "https://pastebin.com/raw/9tcVqgDm",
    [16389398622]     = "https://pastebin.com/raw/eYsUQMtz",
    [16389395869]     = "https://pastebin.com/raw/eYsUQMtz",
    [17527914941]     = "https://pastebin.com/raw/eYsUQMtz",
    [18799085098]     = "https://pastebin.com/raw/T4iwTVKC",

    [77338972879392] = "https://pastebin.com/raw/yEg6npfK",
}

local CurrentPlaceId = game.PlaceId
local TargetScriptUrl = GameScripts[CurrentPlaceId]

-- Check if current game is supported
if not TargetScriptUrl then
    warn("[Nocytra Hub] Game ini belum didukung.")
    return
end

-- Minimal Anti-AFK
local VirtualUser = cloneref(game:GetService("VirtualUser"))

if VirtualUser then
    Players.LocalPlayer.Idled:Connect(function()
        VirtualUser:CaptureController()
        VirtualUser:ClickButton2(Vector2.new())
    end)
end

-- Load and execute the game-specific script
print("[Nocytra Hub] Memuat script untuk PlaceId: " .. CurrentPlaceId)

local success, errorMessage = pcall(function()
    loadstring(game:HttpGet(TargetScriptUrl, true))()
end)

if not success then
    warn("[Nocytra Hub] Gagal memuat script: " .. tostring(errorMessage))
end
